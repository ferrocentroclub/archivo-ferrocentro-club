<script>
    const tableroElemento = document.getElementById('tablero');
    const celdas = document.querySelectorAll('.celda');
    const estadoTexto = document.getElementById('estado-juego');
    const btnReiniciar = document.getElementById('btn-reiniciar');
    
    let tablero = ["", "", "", "", "", "", "", "", ""];
    let juegoActivo = true;

    const combinacionesGanadoras = [
        [0, 1, 2], [3, 4, 5], [6, 7, 8], // Horizontales
        [0, 3, 6], [1, 4, 7], [2, 5, 8], // Verticales
        [0, 4, 8], [2, 4, 6]             // Diagonales
    ];

    // Efecto visual al pasar el ratón por encima de celdas vacías
    celdas.forEach(celda => {
        celda.addEventListener('mouseenter', () => {
            if (tablero[celda.getAttribute('data-i')] === "" && juegoActivo) {
                celda.style.backgroundColor = '#415a77';
            }
        });
        celda.addEventListener('mouseleave', () => {
            if (tablero[celda.getAttribute('data-i')] === "") {
                celda.style.backgroundColor = '#1b263b';
            }
        });
        celda.addEventListener('click', clickCelda);
    });

    btnReiniciar.addEventListener('click', reiniciarJuego);

    function clickCelda(e) {
        const index = e.target.getAttribute('data-i');
        
        if (tablero[index] !== "" || !juegoActivo) return;

        marcarCelda(e.target, index, "X", "#fff");
        e.target.style.backgroundColor = '#1b263b';
        
        if (verificarGanador("X")) {
            finalizarJuego("¡Ganaste! 🎉");
            return;
        }

        if (tablero.includes("")) {
            juegoActivo = false;
            estadoTexto.innerText = "Pensando...";
            setTimeout(turnoCPU, 600);
        } else {
            finalizarJuego("¡Empate! 🤝");
        }
    }

    function turnoCPU() {
        let vacias = [];
        tablero.forEach((val, idx) => { if (val === "") vacias.push(idx); });

        if (vacias.length === 0) return;

        const eleccionAleatoria = vacias[Math.floor(Math.random() * vacias.length)];
        const celdaObjetivo = document.querySelector(`[data-i='${eleccionAleatoria}']`);

        marcarCelda(celdaObjetivo, eleccionAleatoria, "O", "#e9c46a");

        if (verificarGanador("O")) {
            finalizarJuego("¡Perdiste! 💀");
        } else if (!tablero.includes("")) {
            finalizarJuego("¡Empate! 🤝");
        } else {
            juegoActivo = true;
            estadoTexto.innerText = "Tu turno (X)";
        }
    }

    function marcarCelda(celda, index, jugador, color) {
        tablero[index] = jugador;
        celda.innerText = jugador;
        celda.style.color = color;
        celda.style.cursor = "default";
    }

    function verificarGanador(jugador) {
        return combinacionesGanadoras.some(combinacion => {
            return combinacion.every(index => tablero[index] === jugador);
        });
    }

    function finalizarJuego(mensaje) {
        estadoTexto.innerText = mensaje;
        juegoActivo = false;
        btnReiniciar.style.display = "block";
    }

    function reiniciarJuego() {
        tablero = ["", "", "", "", "", "", "", "", ""];
        juegoActivo = true;
        estadoTexto.innerText = "Tu turno (X)";
        btnReiniciar.style.display = "none";
        celdas.forEach(celda => {
            celda.innerText = "";
            celda.style.cursor = "pointer";
            celda.style.backgroundColor = '#1b263b';
        });
    }
</script>
